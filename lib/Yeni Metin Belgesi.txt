function doGet(e) {
  var output = ContentService.createTextOutput();
  output.setMimeType(ContentService.MimeType.JSON);

  try {
    var sheetName = e && e.parameter  e.parameter.sheet  null;
    if (!sheetName) {
      return output.setContent(JSON.stringify({ success false, error Eksik parametre sheet }));
    }

    var ss = SpreadsheetApp.openById(SHEET_ID);
    var sheet = ss.getSheetByName(sheetName);
    if (!sheet) {
      return output.setContent(JSON.stringify({ success false, error Sayfa bulunamadı  + sheetName }));
    }

    var values = sheet.getDataRange().getValues();
    if (values.length  2) {
      return output.setContent(JSON.stringify({ success true, data [] }));
    }

    var action = e && e.parameter  e.parameter.action  null;
    if (action === getNotifications) {
      return responseSuccess(getNotificationsForUser(e.parameter.user_id));
    }

    var headers = values[0];
    var rows = values.slice(1);

    var jsonData = [];

    for (var i = 0; i  rows.length; i++) {
      var obj = {};
      for (var j = 0; j  headers.length; j++) {
        var header = headers[j];
        var value = rows[i][j];

        if (value === undefined  value === null  value === ) {
          obj[header] = ;
        } else if (value instanceof Date) {
          obj[header] = value.toISOString();
        } else if (typeof value === number) {
          obj[header] = value.toString();
        } else {
          obj[header] = value.toString();
        }
      }
      jsonData.push(obj);
    }

    Logger.log(✅  + sheetName +  sayfasından  + jsonData.length +  satır veri çekildi);

    return output.setContent(JSON.stringify({ success true, data jsonData }));

  } catch (err) {
    Logger.log(❌ GET Hatası  + err.toString());
    return output.setContent(JSON.stringify({ success false, error err.toString() }));
  }
}

    var request = JSON.parse(requestBody);
    var action = request.action;
    Logger.log(📥 doPost action  + action);

    if (!action) return responseError(action gerekli.);

     🔒 SADECE yazma action'ları kısa kilit alır. Okumalar (getAllDataBulk,
     login, getNotifications, getImageUrl, getVersionReport, checkMonthlyFees,
     getAssistantsForCoach) kilit görmez - paralel çalışır.
    if (WRITE_ACTIONS[action]) {
      var lock = LockService.getScriptLock();
       tryLock 15 sn içinde sıra gelmezse meşgul dön, client tekrar denesin.
       waitLock gibi 30 sn boyunca thread'i tutup timeout'a düşmez.
      if (!lock.tryLock(15000)) {
        return responseError(BUSY_RETRY);  client bunu görünce backoff'la tekrar denemeli
      }
      try {
        return route(request, action);
      } finally {
        lock.releaseLock();
      }
    }

     Okuma yolu kilitsiz
    return route(request, action);

  } catch (err) {
    Logger.log(❌ doPost hatası  + err.toString());
    return responseError(Sunucu hatası  + err.toString());
  }
}

function handleLogin(email, password) {
  var ss = SpreadsheetApp.openById(SHEET_ID);
  var sheet = ss.getSheetByName(users);
  if (!sheet) return responseError(Users sayfası bulunamadı.);

  var values = sheet.getDataRange().getValues();
  var headers = values[0];

  var emailColIndex = headers.indexOf(email);
  var passColIndex = headers.indexOf(password_hash);
  var lastLoginColIndex = headers.indexOf(last_login);

  var foundUser = null;
  var foundRowIndex = -1;

  for (var i = 1; i  values.length; i++) {
    var sheetEmail = (values[i][emailColIndex]  ).toString().trim();
    var sheetPass = (values[i][passColIndex]  ).toString().trim();
    var reqEmail = (email  ).toString().trim();
    var reqPass = (password  ).toString().trim();

    if (sheetEmail === reqEmail && sheetPass === reqPass) {
      foundUser = {};
      headers.forEach(function(h, idx) {
        foundUser[h] = values[i][idx];
      });
      foundRowIndex = i + 1;
      break;
    }
  }

  if (foundUser) {
    if (lastLoginColIndex !== -1) {
      var now = new Date().toISOString().replace('T', ' ').substring(0, 19);
      sheet.getRange(foundRowIndex, lastLoginColIndex + 1).setValue(now);
      foundUser[last_login] = now;
    }
    foundUser[password_hash] = ;
    return responseSuccess({ user foundUser });
  }

  return responseError(E-posta veya şifre hatalı.);
}

function _attDay(v) {
  if (v instanceof Date) {
    return Utilities.formatDate(v, Session.getScriptTimeZone(), yyyy-MM-dd);
  }
  var s = (v === null  v === undefined)    v.toString().trim();
  if (s.indexOf(T) !== -1) s = s.split(T)[0];
  if (s.indexOf( ) !== -1) s = s.split( )[0];
  return s;
}

function handleSaveBulkAttendance(rows) {
  try {
    if (!rows  !rows.length) return responseError(Boş yoklama listesi.);
 
    var ss = SpreadsheetApp.openById(SHEET_ID);
    var sheet = ss.getSheetByName(attendances);
    if (!sheet) return responseError(attendances sayfası bulunamadı!);
 
    var values = sheet.getDataRange().getValues();
    var headers = values[0];
    var now = new Date().toISOString();
 
    var idCol      = headers.indexOf(attendances_id);
    var studentCol = headers.indexOf(student_id);
    var groupCol   = headers.indexOf(groups_id);
    var dateCol    = headers.indexOf(attendance_date);
    var statusCol  = headers.indexOf(status);
    var takenByCol = headers.indexOf(taken_by);
    var noteCol    = headers.indexOf(note);
    var updatedCol = headers.indexOf(updated_at);
 
    if (studentCol === -1  groupCol === -1  dateCol === -1) {
      return responseError(Gerekli sütunlar bulunamadı!);
    }
 
     Var olan satırları öğrencigrupgün ile indeksle + en büyük ID
    var rowByKey = {};
    var maxId = 0;
    for (var i = 1; i  values.length; i++) {
      var sId = (values[i][studentCol]  ).toString().trim();
      var gId = (values[i][groupCol]    ).toString().trim();
      var dY  = _attDay(values[i][dateCol]);            🔥 Date-güvenli
      if (sId && gId && dY) rowByKey[sId +  + gId +  + dY] = i + 1;
 
      var n = parseInt((values[i][idCol]  ).toString().trim(), 10);
      if (!isNaN(n) && n  maxId) maxId = n;
    }
    var nextId = maxId + 1;
 
    function buildRow(rec) {
      var arr = [];
      for (var h = 0; h  headers.length; h++) {
        arr.push(rec[headers[h]] !== undefined  rec[headers[h]]  );
      }
      return arr;
    }
 
    var newRows = [];
    var pendingIndexByKey = {};
    var updated = 0, inserted = 0;
 
    for (var r = 0; r  rows.length; r++) {
      var d = rows[r]  {};
      var studentId = (d.student_id  ).toString().trim();
      var groupsId  = (d.groups_id   ).toString().trim();
      var attDateRaw = (d.attendance_date  ).toString().trim();
      if (!studentId  !groupsId  !attDateRaw) continue;
 
      var status  = (d.status  ).toString().trim();
      var takenBy = (d.taken_by  ).toString().trim();
      var note    = (d.note  ).toString();
      var key = studentId +  + groupsId +  + _attDay(attDateRaw);   🔥
 
      if (rowByKey[key]) {
         🔁 AYNI GÜN VAR → GÜNCELLE (yeni satır AÇMAZ)
        var rw = rowByKey[key];
        if (statusCol  !== -1) sheet.getRange(rw, statusCol  + 1).setValue(status);
        if (takenByCol !== -1 && takenBy) sheet.getRange(rw, takenByCol + 1).setValue(takenBy);
        if (noteCol    !== -1) sheet.getRange(rw, noteCol + 1).setValue(note);
        if (updatedCol !== -1) sheet.getRange(rw, updatedCol + 1).setValue(now);
        updated++;
 
      } else if (pendingIndexByKey[key] !== undefined) {
        var pi = pendingIndexByKey[key];
        newRows[pi] = buildRow({
          attendances_id newRows[pi][idCol],
          groups_id groupsId, student_id studentId,
          taken_by takenBy, attendance_date attDateRaw,
          status status, note note, updated_at now
        });
 
      } else {
         ➕ YENİ → SIRALI numeric ID
        pendingIndexByKey[key] = newRows.length;
        newRows.push(buildRow({
          attendances_id nextId++,

function getNextId(sheet) {
  var lastRow = sheet.getLastRow();
  if (lastRow  2) return 1;

  var data = sheet.getRange(2, 1, lastRow - 1, 1).getValues();
  var maxId = 0;

  for (var i = 0; i  data.length; i++) {
    var val = data[i][0];
    if (val && val.toString().trim() !== ) {
      var num = parseInt(val.toString().trim());
      if (!isNaN(num) && num  maxId) {
        maxId = num;
      }
    }
  }

  if (maxId === 0) return 1;
  return maxId + 1;
}

function getFCMAccessToken() {
  try {
    var scriptProperties = PropertiesService.getScriptProperties();
    var cachedToken = scriptProperties.getProperty(FCM_ACCESS_TOKEN);
    var tokenExpiry = scriptProperties.getProperty(FCM_TOKEN_EXPIRY);

    if (cachedToken && tokenExpiry) {
      var expiryTime = parseInt(tokenExpiry);
      if (Date.now()  expiryTime) {
        Logger.log(✅ Cached token kullanılıyor);
        return cachedToken;
      }
    }

    var header = {
      alg RS256,
      typ JWT
    };

    var now = Math.floor(Date.now()  1000);
    var claim = {
      iss SERVICE_ACCOUNT.client_email,
      scope httpswww.googleapis.comauthfirebase.messaging,
      aud SERVICE_ACCOUNT.token_uri,
      exp now + 3600,
      iat now
    };

    var encodedHeader = Utilities.base64EncodeWebSafe(JSON.stringify(header));
    var encodedClaim = Utilities.base64EncodeWebSafe(JSON.stringify(claim));
    var toSign = encodedHeader + . + encodedClaim;

    var signature = Utilities.computeRsaSha256Signature(toSign, SERVICE_ACCOUNT.private_key);
    var encodedSignature = Utilities.base64EncodeWebSafe(signature);

    var jwt = toSign + . + encodedSignature;

    var payload = {
      grant_type urnietfparamsoauthgrant-typejwt-bearer,
      assertion jwt
    };

    var options = {
      method post,
      contentType applicationx-www-form-urlencoded,
      payload payload,
      muteHttpExceptions true
    };

    var response = UrlFetchApp.fetch(SERVICE_ACCOUNT.token_uri, options);
    var result = JSON.parse(response.getContentText());

    if (result.access_token) {
      scriptProperties.setProperty(FCM_ACCESS_TOKEN, result.access_token);
      scriptProperties.setProperty(FCM_TOKEN_EXPIRY, (Date.now() + 55  60  1000).toString());
      Logger.log(✅ Yeni access token alındı!);
      return result.access_token;
    }

    throw new Error(Token alınamadı  + JSON.stringify(result));

  } catch(e) {
    Logger.log(❌ Token hatası  + e.toString());
    throw e;
  }
}

  var id = (request.id  ).toString().trim();
  if (!id) return responseError(id gerekli.);

  var ss = SpreadsheetApp.openById(SHEET_ID);
  var sh = ss.getSheetByName(name);
  var values = sh.getDataRange().getValues();
  var headers = values[0];
  var data = request.data  {};

  var rowIndex = -1;
  for (var i = 1; i  values.length; i++) {
    if ((values[i][0]  ).toString().trim() === id) { rowIndex = i + 1; break; }
  }
  if (rowIndex === -1) return responseError(Kayıt yok  + id);

  headers.forEach(function (h, c) {
    if (h === headers[0]) return;  id değişmez
    if (data[h] !== undefined) sh.getRange(rowIndex, c + 1).setValue(data[h]);
  });
  var upC = headers.indexOf(updated_at);
  if (upC !== -1) sh.getRange(rowIndex, upC + 1).setValue(new Date().toISOString());
  return responseSuccess({ message ✅ Güncellendi, id id, table name });
}
